#!/usr/bin/env node
// The Trolls' Hall: "In the Hall of the Mountain King" (Grieg, 1875; public domain) played by the phase harp along ONE
// closed route among the zeros and poles of a rational F (the app's zeros-and-poles step). The route is computed from
// the score (score.js), so the melody, tempo and dynamics can be changed there and the route made again.
//   node make.js     writes route.json (the example's data) and proof.json (what the app's own harp logic plays on it)
//
// How: each crossing k of the score is a knot in the log-F plane, zeta = ln|F| + i arg F, put just past its Schmitt
// threshold (line + dir (H + 0.002 turn)). Consecutive knots are joined by a cubic Bezier in zeta that never reaches
// another threshold (a step: monotone in arg; a turn back over the same line: a hairpin), lifted to z through F by
// Newton continuation. Listen along plays the curve uniformly in arc length, so the arc between two knots must be V
// times the time between the notes: the Bezier bulges in ln|F| (both control points moved by e) and e is found by
// bisection. The far knot's ln|F| follows a register plan (low near the zeros early, high near the poles at the
// climax); when no e works the knot's ln|F| is moved, or the walk backs up. The last chord's knot joins the first
// note's (shifted by W turns) in the same way, so the route closes with no cut and winding number W.
'use strict';
const fs = require('fs'), path = require('path');
const OUT = __dirname;
const SC = require('./score.js');
const { thr, H, NAMES } = SC;
const TAU = 2 * Math.PI, DEL = 0.002;

// ------------------------------------------------------------------ the function: the king's hall
// F(z) = (z - z1)(z - z2)(z - z3) / ((z - p1)(z - p2)(z - p3)(z - K)): three trolls (zeros) and three troll guards
// (poles) alternate on a ring round the Mountain King, a pole at the centre. F -> 0 far away (dark, low), F -> inf at
// the poles (bright, high).
// ---- complex arithmetic on [re, im]
const cm = (a, b) => [a[0] * b[0] - a[1] * b[1], a[0] * b[1] + a[1] * b[0]];
const cdv = (a, b) => { const d = b[0] * b[0] + b[1] * b[1]; return [(a[0] * b[0] + a[1] * b[1]) / d, (a[1] * b[0] - a[0] * b[1]) / d]; };
const cs = (a, b) => [a[0] - b[0], a[1] - b[1]];
const cabs = (a) => Math.hypot(a[0], a[1]);
const cexp = (a) => { const r = Math.exp(a[0]); return [r * Math.cos(a[1]), r * Math.sin(a[1])]; };
const clog = (a) => [Math.log(cabs(a)), Math.atan2(a[1], a[0])];
function makeF(isl) {
  const zs = isl.zeros, ps = isl.poles, deg = Math.max(zs.length, ps.length), a = isl.a;
  const F = (z) => {
    let w = a;
    for (let i = 0; i < deg; i++) {
      if (i < zs.length && i < ps.length) w = cm(w, cdv(cs(z, zs[i]), cs(z, ps[i])));
      else if (i < zs.length) w = cm(w, cs(z, zs[i]));
      else w = cdv(w, cs(z, ps[i]));
    }
    return w;
  };
  const g = (z) => {   // F'/F
    let r = [0, 0];
    for (const q of zs) { const v = cdv([1, 0], cs(z, q)); r = [r[0] + v[0], r[1] + v[1]]; }
    for (const q of ps) { const v = cdv([1, 0], cs(z, q)); r = [r[0] - v[0], r[1] - v[1]]; }
    return r;
  };
  return { F, g };
}

// ------------------------------------------------------------------ lifting a zeta path through F
// solve F(z) = e^zeta near z (Newton on log F - zeta, principal: the steps are small)
function solve(FG, z, zeta) {
  for (let it = 0; it < 30; it++) {
    const r = clog(cm(FG.F(z), cexp([-zeta[0], -zeta[1]])));
    if (cabs(r) < 1e-13) return z;
    const dz = cdv(r, FG.g(z));
    const s = Math.min(1, 0.25 / Math.max(1e-12, cabs(dz)));   // damped near a critical point
    z = [z[0] - s * dz[0], z[1] - s * dz[1]];
  }
  const r = clog(cm(FG.F(z), cexp([-zeta[0], -zeta[1]])));
  return cabs(r) < 1e-9 ? z : null;
}
const bez = (P, u) => { const v = 1 - u, a = v * v * v, b = 3 * v * v * u, c = 3 * v * u * u, d = u * u * u; return [a * P[0][0] + b * P[1][0] + c * P[2][0] + d * P[3][0], a * P[0][1] + b * P[1][1] + c * P[2][1] + d * P[3][1]]; };
// the lift of the Bezier P (zeta) from z0: points (the first is z0) and length; null when it fails
function lift(FG, P, z0, maxDz = 0.01) {
  const pts = [z0];
  let z = z0, u = 0, h = 1 / 256, L = 0;
  while (u < 1) {
    const u1 = Math.min(1, u + h), zt = bez(P, u1), gz = FG.g(z), guess = [z[0], z[1]];
    // predictor: dz = dzeta / g
    const zp = bez(P, u), dzeta = [zt[0] - zp[0], zt[1] - zp[1]], pr = cdv(dzeta, gz);
    const z1 = solve(FG, [guess[0] + pr[0], guess[1] + pr[1]], zt);
    if (!z1 || cabs(cs(z1, z)) > maxDz || cabs(cs(z1, [guess[0] + pr[0], guess[1] + pr[1]])) > 0.2 * cabs(pr) + 1e-9) {
      h /= 2; if (h < 1e-9) return null; continue;
    }
    L += cabs(cs(z1, z)); z = z1; u = u1; pts.push(z);
    if (cabs(cs(z1, z)) < 0.3 * maxDz) h = Math.min(h * 1.5, 1 / 64);
  }
  return { pts, L };
}

const ring = (r, deg) => [r * Math.cos(deg * Math.PI / 180), r * Math.sin(deg * Math.PI / 180)];
// six trolls (zeros) on a ring of radius 1.6 round the Mountain King (a pole at 0): F = a (z^6 - 1.6^6) / z. Inside
// the ring F ~ a 1.6^6 / z: arg F turns once clockwise round the King, |F| grows toward him (the register rises)
const HALL = {
  a: [0.01, 0],
  zeros: [0, 60, 120, 180, 240, 300].map((d) => ring(1.6, d + 30)),
  poles: [[0, 0]],
};
const ZMAX = 1.05;   // the route stays inside the hall (the critical points of F are at |z| = 1.6 / 5^(1/6) = 1.23)
const VDEF = 3;   // the route's speed, plane units a second (Listen along: uniform in arc length)

// ------------------------------------------------------------------ the register plan: ln|F| at each note
// low and dark early, rising with the crescendo; inside a statement the turning notes go deeper (near a troll), the
// chords sit close to the King
function plan(e) {
  const S = SC, u = Math.min(1, Math.max(0, (e.t - S.LEAD) / (S.timeOf(S.BEATS) - S.LEAD)));
  const rho = e.kind === 'chord' ? 0.022 : 0.8 * Math.pow(0.12 / 0.8, u);   // distance from the King
  return Math.log(0.01 * Math.pow(1.6, 6) / rho);
}
function build(o = {}) {
  const FG = makeF(o.hall || HALL), V = o.V || VDEF;
  const S = SC.score(o);
  const ev = S.ev, K = ev.length, W = S.W;
  for (const e of ev) { e.Th = TAU * (thr(e.line) + e.dir * (H + DEL)); e.rp = plan(e); }
  const adj = o.adj || new Array(K + 1).fill(0);
  const band = (c) => thr(c + 1) - thr(c);
  function bezier(na, nb, ra, rb, bulge) {
    const A = [ra, na.Th], B = [rb, nb.Th];
    if (na.dir === nb.dir) {
      const h = 0.42 * Math.abs(nb.Th - na.Th);
      return [A, [ra + bulge, na.Th + na.dir * h], [rb + bulge, nb.Th - nb.dir * h], B];
    }
    // a turn back over the same line: a hairpin out past it into the cell and back
    const c = na.dir > 0 ? na.line : na.line - 1, short = nb.kind === 'flam' || na.kind === 'chord' || nb.kind === 'chord';
    const h = TAU * (short ? 0.012 : 0.22 * band(c));
    return [A, [ra + bulge, na.Th + na.dir * h], [rb + bulge, nb.Th + na.dir * h], B];
  }
  const nf = Object.assign({}, ev[0], { Th: ev[0].Th + TAU * W });
  const nextOf = (k) => (k + 1 < K ? ev[k + 1] : nf);
  const durOf = (k) => (k + 1 < K ? ev[k + 1].t - ev[k].t : S.tEnd - ev[k].t + ev[0].t);
  // seat the first knot
  const r0 = ev[0].rp;
  // (the seat: the grid point nearest the first knot, preferring o.z0's side, then Newton)
  let z0 = null;
  { let best = Infinity, g0 = null; const pref = o.z0 || [2.4, 0.6];
    for (let x = -4; x <= 4; x += 0.05) for (let y = -4; y <= 4; y += 0.05) {
      const l = clog(cm(FG.F([x, y]), cexp([-r0, -ev[0].Th]))), d = cabs(l) + 0.02 * Math.hypot(x - pref[0], y - pref[1]);
      if (d < best) { best = d; g0 = [x, y]; } }
    z0 = solve(FG, g0, [r0, ev[0].Th]); }
  if (!z0) throw new Error('cannot seat the first note');
  function interval(k, kn, sgn) {
    const na = ev[k], nb = nextOf(k), ra = kn.r, want = V * (durOf(k) + adj[k]);
    const last = k + 1 === K;
    const pr = last ? r0 : nb.rp;
    const inHall = (l) => l && l.pts.every((p) => p[0] * p[0] + p[1] * p[1] < (o.zmax || ZMAX) ** 2);
    const lenOf = (rb, e) => { const l = lift(FG, bezier(na, nb, ra, rb, e), kn.z); return inHall(l) ? l.L : NaN; };
    const deep = Math.sign(pr) || 1;
    const cands = last ? [r0] : [pr, pr + deep * 0.8, pr + deep * 1.6, pr - deep * 0.8, 0.5 * (pr + ra), ra];
    for (const rb of cands) {
      const L0 = lenOf(rb, 0);
      if (!(L0 <= want)) continue;
      let emax = 0.4, Lm = lenOf(rb, sgn * emax);
      while (Lm < want && emax < 6) { emax *= 1.5; Lm = lenOf(rb, sgn * emax); }
      if (!(Lm >= want)) continue;
      let lo = 0, hi = emax;
      for (let it = 0; it < 48; it++) { const m = 0.5 * (lo + hi), lm = lenOf(rb, sgn * m); if (lm < want) lo = m; else hi = m; }
      const e = sgn * 0.5 * (lo + hi), P = bezier(na, nb, ra, rb, e), l = lift(FG, P, kn.z);
      if (!inHall(l)) continue;
      return { knot: { z: l.pts[l.pts.length - 1], r: rb }, seg: { pts: l.pts, L: l.L, P, e, k } };
    }
    return null;
  }
  const knots = [{ z: z0, r: r0 }], segs = [], tried = [];
  let k = 0, steps = 0;
  while (k < K) {
    if (++steps > (o.maxSteps || 3000)) throw new Error(`no route found (stuck near event ${k})`);
    const pref = (o.sign && o.sign[k]) || (knots[k].r >= 0 ? -1 : 1);
    tried[k] = tried[k] || [];
    let got = null;
    for (const sgn of [pref, -pref]) {
      if (tried[k].includes(sgn)) continue;
      tried[k].push(sgn);
      got = interval(k, knots[k], sgn);
      if (got) break;
    }
    if (process.env.DBG) console.log('k', k, ev[k].kind, ev[k].nm, 'got', !!got, knots[k].r.toFixed(2), got ? got.seg.e.toFixed(2) : '');
    if (got) { knots[k + 1] = got.knot; segs[k] = got.seg; k++; tried[k] = []; continue; }
    if (k === 0) throw new Error('no route from the first note');
    tried[k] = null; k--;
  }
  const gap = cabs(cs(knots[K].z, knots[0].z));
  return { FG, V, hall: o.hall || HALL, S, ev, knots, segs, gap, total: segs.reduce((s, q) => s + q.L, 0) };
}

// the route as a closed polyline (the dense lifted points), starting LEAD before the first note
function polyline(R) {
  const dense = [];
  for (const s of R.segs) for (let i = dense.length ? 1 : 0; i < s.pts.length; i++) dense.push(s.pts[i]);
  dense.pop();   // the last point is the first knot again
  const cum = [0];
  for (let i = 1; i < dense.length; i++) cum.push(cum[i - 1] + cabs(cs(dense[i], dense[i - 1])));
  const L = cum[cum.length - 1] + cabs(cs(dense[0], dense[dense.length - 1])), s0 = L - R.V * R.ev[0].t;
  // rotate so that the polyline starts at arc s0 (inserting that point)
  let j = 0; while (j < cum.length - 1 && cum[j + 1] <= s0) j++;
  const a = dense[j], b = dense[(j + 1) % dense.length], seg = cabs(cs(b, a)), u = (s0 - cum[j]) / seg;
  const p0 = [a[0] + (b[0] - a[0]) * u, a[1] + (b[1] - a[1]) * u];
  const out = [p0];
  for (let i = j + 1; i < dense.length; i++) out.push(dense[i]);
  for (let i = 0; i <= j; i++) out.push(dense[i]);
  return { pts: out, L };
}

module.exports = { HALL, VDEF, build, polyline, makeF, plan };

if (require.main === module) {
  const t0 = Date.now();
  const R = build();
  const P = polyline(R);
  console.log('events', R.ev.length, 'gap', R.gap.toExponential(2), 'length', R.total.toFixed(3), 'dur', (R.total / R.V).toFixed(3), 'pts', P.pts.length, 'ms', Date.now() - t0);
  const r5 = (x) => Math.round(x * 1e5) / 1e5;
  fs.writeFileSync(path.join(OUT, 'route.json'), JSON.stringify({ hall: R.hall, V: R.V, dur: R.total / R.V, W: R.S.W, closed: true, pts: P.pts.map((p) => [r5(p[0]), r5(p[1])]) }));
  if (process.argv.includes('--png')) sketch(R, path.join(OUT, 'sketch.png'));
}
// ------------------------------------------------------------------ a rough sketch (not the app's rendering)
function sketch(R, file, box) {
  const zlib = require('zlib'), Wd = 720, Ht = 520;
  const pts = R.segs.flatMap((s) => s.pts);
  let [x0, x1, y0, y1] = box || [Infinity, -Infinity, Infinity, -Infinity];
  if (!box) { for (const p of pts) { x0 = Math.min(x0, p[0]); x1 = Math.max(x1, p[0]); y0 = Math.min(y0, p[1]); y1 = Math.max(y1, p[1]); } }
  const cx = (x0 + x1) / 2, cy = (y0 + y1) / 2, sc = 0.85 * Math.min(Wd / (x1 - x0 || 1), Ht / (y1 - y0 || 1));
  const img = Buffer.alloc(Wd * Ht * 3);
  const TH5 = [0, 1 / 6, 1 / 3, 7 / 12, 3 / 4, 1];
  const COL = [[230, 120, 110], [235, 190, 100], [150, 200, 120], [110, 170, 220], [175, 140, 215]];
  for (let py = 0; py < Ht; py++) for (let px = 0; px < Wd; px++) {
    const z = [cx + (px - Wd / 2) / sc, cy - (py - Ht / 2) / sc], w = R.FG.F(z);
    let t = Math.atan2(w[1], w[0]) / (2 * Math.PI); t -= Math.floor(t);
    let b = 0; while (t >= TH5[b + 1]) b++;
    const lr = Math.log(Math.hypot(w[0], w[1])), shade = 0.82 + 0.18 * (lr - Math.floor(lr));
    const o = (py * Wd + px) * 3;
    for (let c = 0; c < 3; c++) img[o + c] = Math.min(255, COL[b][c] * shade);
  }
  const dot = (p, r, col) => { const X = Wd / 2 + (p[0] - cx) * sc, Y = Ht / 2 - (p[1] - cy) * sc; for (let dy = -r; dy <= r; dy++) for (let dx = -r; dx <= r; dx++) { if (dx * dx + dy * dy > r * r) continue; const xx = Math.round(X + dx), yy = Math.round(Y + dy); if (xx < 0 || yy < 0 || xx >= Wd || yy >= Ht) continue; const o = (yy * Wd + xx) * 3; img[o] = col[0]; img[o + 1] = col[1]; img[o + 2] = col[2]; } };
  for (let i = 1; i < pts.length; i++) for (let u = 0; u < 1; u += 0.2) dot([pts[i - 1][0] + u * (pts[i][0] - pts[i - 1][0]), pts[i - 1][1] + u * (pts[i][1] - pts[i - 1][1])], 1, [20, 20, 30]);
  for (const k of R.knots) dot(k.z, 3, [255, 255, 255]);
  const isl = R.hall;
  for (const z of isl.zeros) dot(z, 5, [0, 0, 0]);
  for (const p of isl.poles) dot(p, 5, [255, 255, 255]);
  const raw = Buffer.alloc((Wd * 3 + 1) * Ht);
  for (let y = 0; y < Ht; y++) { raw[y * (Wd * 3 + 1)] = 0; img.copy(raw, y * (Wd * 3 + 1) + 1, y * Wd * 3, (y + 1) * Wd * 3); }
  const crc = (buf) => { let c = ~0; for (const b of buf) { c ^= b; for (let k = 0; k < 8; k++) c = (c >>> 1) ^ (0xEDB88320 & -(c & 1)); } return ~c >>> 0; };
  const chunk = (type, data) => { const len = Buffer.alloc(4); len.writeUInt32BE(data.length); const td = Buffer.concat([Buffer.from(type), data]); const cr = Buffer.alloc(4); cr.writeUInt32BE(crc(td)); return Buffer.concat([len, td, cr]); };
  const ihdr = Buffer.alloc(13); ihdr.writeUInt32BE(Wd, 0); ihdr.writeUInt32BE(Ht, 4); ihdr[8] = 8; ihdr[9] = 2;
  fs.writeFileSync(file, Buffer.concat([Buffer.from([137, 80, 78, 71, 13, 10, 26, 10]), chunk('IHDR', ihdr), chunk('IDAT', zlib.deflateSync(raw)), chunk('IEND', Buffer.alloc(0))]));
}
module.exports.sketch = sketch;
